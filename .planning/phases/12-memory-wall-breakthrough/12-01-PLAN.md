</read_first>

<action>
1. Em cada kernel `batch_lookup_*`, substituir o prefetching de distância 1 por um pipeline de 3 estágios com distância 8:

**Padrão a aplicar em `batch_lookup_avx2` (e equivalente nos outros):**
```cpp
void batch_lookup_avx2(const uint16_t* buckets, size_t mask, const uint16_t* fps,
                        const size_t* i1s, const size_t* i2s, bool* results, size_t count) {
    constexpr size_t kPrefetchDist = 8;

    // Estágio 1: Pré-aquecer prefetch para os primeiros kPrefetchDist elementos
    for (size_t i = 0; i < std::min(kPrefetchDist, count); ++i) {
        _mm_prefetch(reinterpret_cast<const char*>(&buckets[i1s[i] * 4]), _MM_HINT_T0);
        _mm_prefetch(reinterpret_cast<const char*>(&buckets[i2s[i] * 4]), _MM_HINT_T0);
    }

    // Estágio 2+3: Processar com prefetch avançado
    for (size_t i = 0; i < count; ++i) {
        // Dispara prefetch para i + kPrefetchDist (futuro)
        if (i + kPrefetchDist < count) {
            _mm_prefetch(reinterpret_cast<const char*>(&buckets[i1s[i + kPrefetchDist] * 4]), _MM_HINT_T0);
            _mm_prefetch(reinterpret_cast<const char*>(&buckets[i2s[i + kPrefetchDist] * 4]), _MM_HINT_T0);
        }
        results[i] = lookup_avx2(buckets, mask, fps[i], i1s[i], i2s[i]);
    }
}
```

__builtin_prefetch(&buckets[i1s[i + kPrefetchDist] * 4], 0, 3);
__builtin_prefetch(&buckets[i2s[i + kPrefetchDist] * 4], 0, 3);
```

#if defined(__AVX512F__)
{
    const __m512i mask16 = _mm512_set1_epi64(0xFFFFULL);
    const __m512i zero = _mm512_setzero_si512();
    for (size_t i = 0; i < kFleetSize; i += 8) {
        __m512i x0 = _mm512_loadu_si512(&local_fleet.x[0][i]);
        __m512i masked = _mm512_and_si512(x0, mask16);
        __mmask8 hits = _mm512_cmpeq_epi64_mask(masked, zero);
        if (hits == 0) continue; // Fast path: nenhum DP neste bloco de 8
        // Slow path: extrair posições dos bits ativos
        while (hits) {
            int bit = __builtin_ctz(hits);
            size_t idx = i + bit;
            // ... processamento individual (mesmo código atual das linhas 694+)
            hits &= hits - 1;
        }
    }
}
#elif defined(__AVX2__)
{
    const __m256i mask16 = _mm256_set1_epi64x(0xFFFFULL);
    const __m256i zero = _mm256_setzero_si256();
    for (size_t i = 0; i < kFleetSize; i += 4) {
        __m256i x0 = _mm256_loadu_si256((__m256i*)&local_fleet.x[0][i]);
        __m256i masked = _mm256_and_si256(x0, mask16);
        __m256i cmp = _mm256_cmpeq_epi64(masked, zero);
        int movmask = _mm256_movemask_epi8(cmp);
        if (movmask == 0) continue;
        // Extrair posições: cada lane = 8 bytes no movemask
        for (int lane = 0; lane < 4; ++lane) {
            if (movmask & (0xFF << (lane * 8))) {
                size_t idx = i + lane;
                // ... processamento individual
            }
        }
    }
}
#else
// Fallback escalar (código original)
for (size_t i = 0; i < kFleetSize; ++i) {
    if ((local_fleet.x[0][i] & 0xFFFFULL) == 0) {
        // ... processamento individual
    }
}
#endif
```

```cpp
// Em system/hardware.cpp:
#ifdef __linux__
#include <sys/mman.h>
void* allocate_huge_pages(size_t size_bytes) {
    // Round up to 2MB boundary
    size_t aligned = (size_bytes + (2*1024*1024 - 1)) & ~(2*1024*1024 - 1);
    void* ptr = mmap(nullptr, aligned, PROT_READ | PROT_WRITE,
                     MAP_PRIVATE | MAP_ANONYMOUS | MAP_HUGETLB, -1, 0);
    if (ptr == MAP_FAILED) {
        // Fallback: alocação normal com madvise HUGEPAGE hint
        ptr = mmap(nullptr, aligned, PROT_READ | PROT_WRITE,
                   MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);
        if (ptr != MAP_FAILED) {
            madvise(ptr, aligned, MADV_HUGEPAGE);
        }
    }
    return (ptr == MAP_FAILED) ? nullptr : ptr;
}

void free_huge_pages(void* ptr, size_t size_bytes) {
    size_t aligned = (size_bytes + (2*1024*1024 - 1)) & ~(2*1024*1024 - 1);
    munmap(ptr, aligned);
}
#elif defined(_WIN32)
#include <windows.h>
void* allocate_huge_pages(size_t size_bytes) {
    // Tenta MEM_LARGE_PAGES (requer SeLockMemoryPrivilege)
    void* ptr = VirtualAlloc(nullptr, size_bytes,
                             MEM_COMMIT | MEM_RESERVE | MEM_LARGE_PAGES, PAGE_READWRITE);
    if (!ptr) {
        // Fallback: alocação normal
        ptr = VirtualAlloc(nullptr, size_bytes,
                           MEM_COMMIT | MEM_RESERVE, PAGE_READWRITE);
    }
    return ptr;
}

void free_huge_pages(void* ptr, size_t size_bytes) {
    (void)size_bytes;
    VirtualFree(ptr, 0, MEM_RELEASE);
}
#else
// Fallback para plataformas sem suporte
void* allocate_huge_pages(size_t size_bytes) {
    return std::aligned_alloc(64, size_bytes);
}
void free_huge_pages(void* ptr, size_t) {
    std::free(ptr);
}
#endif
```
